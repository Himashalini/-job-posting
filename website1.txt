-- =========================================================
-- DATABASE
-- =========================================================
CREATE DATABASE IF NOT EXISTS job_affiliate;
USE job_affiliate;

-- =========================================================
-- PRIVATE JOBS
-- =========================================================
CREATE TABLE IF NOT EXISTS jobs (
  id INT NOT NULL PRIMARY KEY,
  title VARCHAR(255),
  company VARCHAR(255),
  description TEXT,
  image_url TEXT,
  apply_link TEXT,
  job_type VARCHAR(50)
);

-- =========================================================
-- GOVERNMENT JOBS
-- =========================================================
CREATE TABLE IF NOT EXISTS govt_jobs (
  id INT NOT NULL PRIMARY KEY,
  title VARCHAR(255),
  department VARCHAR(255),
  organization VARCHAR(255),
  location VARCHAR(255),
  qualification VARCHAR(255),
  age_limit VARCHAR(100),
  employment_type VARCHAR(100),
  application_start DATE,
  application_end DATE,
  exam_date DATE,
  salary VARCHAR(100),
  description TEXT,
  apply_link TEXT,
  notification_pdf TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- =========================================================
-- DRESSES (DIRECT BUY + AFFILIATE READY)
-- =========================================================
CREATE TABLE IF NOT EXISTS dresses (
  id INT NOT NULL PRIMARY KEY,
  name VARCHAR(255),
  price DECIMAL(10,2),
  description TEXT,
  type ENUM('direct','partner') NOT NULL DEFAULT 'direct',
  affiliate_link VARCHAR(2048)
);

CREATE TABLE IF NOT EXISTS dress_images (
  id INT AUTO_INCREMENT PRIMARY KEY,
  dress_id INT NOT NULL,
  image_url TEXT,
  FOREIGN KEY (dress_id) REFERENCES dresses(id) ON DELETE CASCADE
);

-- =========================================================
-- JEWELLERY (DIRECT BUY + AFFILIATE READY)
-- =========================================================
CREATE TABLE IF NOT EXISTS jewellery (
  id INT NOT NULL PRIMARY KEY,
  name VARCHAR(255),
  price DECIMAL(10,2),
  description TEXT,
  type ENUM('direct','partner') NOT NULL DEFAULT 'direct',
  affiliate_link VARCHAR(2048)
);

CREATE TABLE IF NOT EXISTS jewellery_images (
  id INT AUTO_INCREMENT PRIMARY KEY,
  jewellery_id INT NOT NULL,
  image_url TEXT,
  FOREIGN KEY (jewellery_id) REFERENCES jewellery(id) ON DELETE CASCADE
);

-- =========================================================
-- BOOKS (AFFILIATE READY)
-- =========================================================
CREATE TABLE IF NOT EXISTS books (
  id INT NOT NULL PRIMARY KEY,
  title VARCHAR(255),
  price DECIMAL(10,2),
  description TEXT,
  type ENUM('direct','partner') NOT NULL DEFAULT 'direct',
  affiliate_link VARCHAR(2048)
);

CREATE TABLE IF NOT EXISTS book_images (
  id INT AUTO_INCREMENT PRIMARY KEY,
  book_id INT NOT NULL,
  image_url TEXT,
  FOREIGN KEY (book_id) REFERENCES books(id) ON DELETE CASCADE
);

-- =========================================================
-- FEEDBACK
-- =========================================================
CREATE TABLE IF NOT EXISTS feedback (
  id INT AUTO_INCREMENT PRIMARY KEY,
  message TEXT
);

-- =========================================================
-- OPTIONAL BUT RECOMMENDED INDEXES
-- =========================================================
CREATE INDEX IF NOT EXISTS idx_dresses_type ON dresses(type);
CREATE INDEX IF NOT EXISTS idx_jewellery_type ON jewellery(type);
CREATE INDEX IF NOT EXISTS idx_books_type ON books(type);
